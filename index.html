01 / EXPRESS & SEMVER · 5 MIN

Enciende el motor

MOTOR

Express procesa las peticiones HTTP y SemVer comunica la versión de la API en cada respuesta mediante el encabezado X-API-Version.

Cliente→Express→Header SemVer→200 OK
PASO 1

Crea api/src/app.js con Express, soporte para JSON, cookies, header SemVer global y ruta de diagnóstico /health.

CREA · api/src/app.js 
1import express from "express"; // Importa el framework Express.
2import cookieParser from "cookie-parser"; // Habilita lectura de cookies en req.cookies.
3import { juegoRouter } from "./routes/juego.routes.js"; // Importa el enrutador de juegos.
4import { manejarError, rutaNoEncontrada } from "./middlewares/error.middleware.js";
5
6const API_VERSION = "1.0.0"; // Versión SemVer (MAJOR.MINOR.PATCH).
7const app = express();
8
9app.use(express.json()); // Parsea el cuerpo de la petición como JSON.
10app.use(cookieParser()); // Parsea el header Cookie.
11
12app.use((req, res, next) => { // Middleware global para versionado SemVer.
13  res.set("X-API-Version", API_VERSION);
14  next();
15});
16
17app.get("/health", (req, res) => { // Diagnóstico de salud del servicio.
18  res.status(200).json({ estado: "ok", version: API_VERSION, servicio: "api-videojuegos" });
19});
20
21app.get("/api/v1/preferencia", (req, res) => { // Lee la cookie enviada por el navegador.
22  res.json({ ultimaPlataforma: req.cookies.ultimaPlataforma ?? null });
23});
24
25app.use("/api/v1/juegos", juegoRouter); // Conecta las rutas CRUD bajo /api/v1/juegos.
26
27app.use(rutaNoEncontrada); // 404 para cualquier ruta inexistente.
28app.use(manejarError); // Manejador central de excepciones.
29
30export default app;

02 / DTO · 4 MIN

Filtra los datos como un escudo

ESCUDO DTO

Un DTO (Data Transfer Object) garantiza que solo los atributos permitidos ingresen a tu aplicación. Si un usuario malicioso envía esAdmin: true o cheatCode: "god_mode", el DTO los descarta en el acto.

BODY RECIBIDO (CON CAMPOS PELIGROSOS) { "titulo": " Elden Ring ", "horasJugadas": "110", "cheatCode": "unlimited" }
OBJETO DTO LIMPIO (SOLO CAMPOS PERMITIDOS) { "titulo": "Elden Ring", "horasJugadas": 110 }
PASO 3

Crea api/src/dtos/juego.dto.js definiendo CrearJuegoDTO y ActualizarJuegoDTO.

CREA · api/src/dtos/juego.dto.js 
1export class CrearJuegoDTO {
2  constructor(data) {
3    this.titulo = data.titulo?.trim(); // Elimina espacios innecesarios al inicio y final.
4    this.genero = data.genero?.trim().toLowerCase(); // Estandariza a minúsculas.
5    this.plataforma = data.plataforma?.trim().toLowerCase();
6    this.horasJugadas = Number(data.horasJugadas ?? 0); // Casteo numérico seguro.
7    this.completado = Boolean(data.completado ?? false);
8    // Cualquier otro campo enviado en data es ignorado por completo.
9  }
10}
11
12export class ActualizarJuegoDTO {
13  constructor(data) {
14    if (data.titulo !== undefined) this.titulo = data.titulo?.trim();
15    if (data.genero !== undefined) this.genero = data.genero?.trim().toLowerCase();
16    if (data.plataforma !== undefined) this.plataforma = data.plataforma?.trim().toLowerCase();
17    if (data.horasJugadas !== undefined) this.horasJugadas = Number(data.horasJugadas);
18    if (data.completado !== undefined) this.completado = Boolean(data.completado);
19  }
20}

VALIDADOR

express-validator actúa antes que el controlador. Si un campo no cumple las reglas, detiene la petición inmediatamente con código 400 Bad Request.

PLATAFORMAS 4 pc, playstation, xbox, nintendo
GÉNEROS 5 accion, rpg, aventura, estrategia, plataformas

Crea api/src/validators/juego.validator.js con las reglas para crear, actualizar y validar el ID de MongoDB.

CREA · api/src/validators/juego.validator.js 
1import { body, param, validationResult } from "express-validator";
2
3const responderErrores = (req, res, next) => {
4  const errores = validationResult(req);
5  if (!errores.isEmpty()) {
6    return res.status(400).json({ errores: errores.array() });
7  }
8  next();
9};
10
11export const GENEROS_VALIDOS = ["accion", "rpg", "aventura", "estrategia", "plataformas"];
12export const PLATAFORMAS_VALIDAS = ["pc", "playstation", "xbox", "nintendo"];
13
14export const validarCrearJuego = [
15  body("titulo").trim().isLength({ min: 2, max: 60 }).withMessage("El título debe tener entre 2 y 60 caracteres"),
16  body("genero").trim().toLowerCase().isIn(GENEROS_VALIDOS).withMessage(`El género debe ser uno de: ${GENEROS_VALIDOS.join(", ")}`),
17  body("plataforma").trim().toLowerCase().isIn(PLATAFORMAS_VALIDAS).withMessage(`La plataforma debe ser una de: ${PLATAFORMAS_VALIDAS.join(", ")}`),
18  body("horasJugadas").optional().isInt({ min: 0 }).withMessage("Las horas jugadas deben ser un número entero mayor o igual a 0"),
19  body("completado").optional().isBoolean().withMessage("El estado completado debe ser un valor booleano"),
20  responderErrores,
21];
22
23export const validarActualizarJuego = [
24  body("titulo").optional().trim().isLength({ min: 2, max: 60 }).withMessage("El título debe tener entre 2 y 60 caracteres"),
25  body("genero").optional().trim().toLowerCase().isIn(GENEROS_VALIDOS).withMessage(`El género debe ser uno de: ${GENEROS_VALIDOS.join(", ")}`),
26  body("plataforma").optional().trim().toLowerCase().isIn(PLATAFORMAS_VALIDAS).withMessage(`La plataforma debe ser una de: ${PLATAFORMAS_VALIDAS.join(", ")}`),
27  body("horasJugadas").optional().isInt({ min: 0 }).withMessage("Las horas jugadas deben ser un número entero mayor o igual a 0"),
28  body("completado").optional().isBoolean().withMessage("El estado completado debe ser un valor booleano"),
29  responderErrores,
30];
31
32export const validarJuegoId = [
33  param("id").isMongoId().withMessage("El id del videojuego no es un ObjectId de MongoDB válido"),
34  responderErrores,
35];

CREA · api/src/models/juego.model.js 
1import { Schema, model } from "mongoose";
2
3const juegoSchema = new Schema({
4  titulo: { type: String, required: true, trim: true, minlength: 2, maxlength: 60 },
5  genero: { type: String, required: true, enum: ["accion", "rpg", "aventura", "estrategia", "plataformas"] },
6  plataforma: { type: String, required: true, enum: ["pc", "playstation", "xbox", "nintendo"] },
7  horasJugadas: { type: Number, default: 0, min: 0 },
8  completado: { type: Boolean, default: false },
9}, { timestamps: true, versionKey: false }); // Añade createdAt y updatedAt automáticamente.
10
11export const Juego = model("Juego", juegoSchema);

2
3export const juegoService = {
4  async crear(datos) { return Juego.create(datos); },
5  async listar(filtro = {}) { return Juego.find(filtro).sort({ createdAt: -1 }); },
6  async buscarPorId(id) { return Juego.findById(id); },
7  async actualizar(id, datos) { return Juego.findByIdAndUpdate(id, datos, { new: true, runValidators: true }); },

9};

PASO 8

Crea api/src/controllers/juego.controller.js. En crearJuego escribe la cookie ultimaPlataforma.

CREA · api/src/controllers/juego.controller.js 
1import { CrearJuegoDTO, ActualizarJuegoDTO } from "../dtos/juego.dto.js";
2import { juegoService } from "../services/juego.service.js";
3
4export async function crearJuego(req, res, next) {
5  try {
6    const datos = new CrearJuegoDTO(req.body);
7    const juego = await juegoService.crear(datos);
8
9    res.cookie("ultimaPlataforma", juego.plataforma, {
10      httpOnly: true,
11      sameSite: "lax",
12      maxAge: 1000 * 60 * 60 * 24, // Recuerda la consola 24 horas.
13    });
14
15    res.status(201).json({ data: juego });
16  } catch (error) { next(error); }
17}
18
19export async function listarJuegos(req, res, next) {
20  try {
21    const filtro = {};
22    if (req.query.plataforma) filtro.plataforma = req.query.plataforma.toLowerCase();
23    if (req.query.genero) filtro.genero = req.query.genero.toLowerCase();
24    const juegos = await juegoService.listar(filtro);
25    res.json({ data: juegos });
26  } catch (error) { next(error); }
27}
28
29export async function obtenerJuego(req, res, next) {
30  try {
31    const juego = await juegoService.buscarPorId(req.params.id);
32    if (!juego) return res.status(404).json({ error: "Videojuego no encontrado" });
33    res.json({ data: juego });
34  } catch (error) { next(error); }
35}
36
37export async function actualizarJuego(req, res, next) {
38  try {
39    const datos = new ActualizarJuegoDTO(req.body);
40    if (Object.keys(datos).length === 0) {
41      return res.status(400).json({ error: "Debes enviar al menos un campo válido para actualizar" });
42    }
43    const juego = await juegoService.actualizar(req.params.id, datos);
44    if (!juego) return res.status(404).json({ error: "Videojuego no encontrado" });
45    res.json({ data: juego });
46  } catch (error) { next(error); }
47}
48
49export async function eliminarJuego(req, res, next) {
50  try {
51    const juego = await juegoService.eliminar(req.params.id);
52    if (!juego) return res.status(404).json({ error: "Videojuego no encontrado" });
53    res.status(204).send(); // 204 No Content: éxito sin cuerpo.
54  } catch (error) { next(error); }
55}
